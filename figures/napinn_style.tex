% 3-D slab.  #1 x  #2 y  #3 w  #4 h  #5 depth  #6 colour
\newcommand{\cube}[6]{%
  \begin{scope}[shift={(#1,#2)}]
    \filldraw[fill=#6!82!black, draw=black!60, line width=0.35pt]
      (0,#4) -- (#5,#4+#5) -- (#3+#5,#4+#5) -- (#3,#4) -- cycle;
    \filldraw[fill=#6!66!black, draw=black!60, line width=0.35pt]
      (#3,0) -- (#3+#5,#5) -- (#3+#5,#4+#5) -- (#3,#4) -- cycle;
    \filldraw[fill=#6, draw=black!60, line width=0.35pt] (0,0) rectangle (#3,#4);
  \end{scope}}
\newcommand{\stack}[5]{\foreach \k in {2,1,0}{\cube{#1+0.15*\k}{#2+0.10*\k}{#3}{#4}{0.14}{#5}}}

\newcommand{\framepanel}[4]{%
  \fill[#3, rounded corners=3pt, draw=black!45, dashed,
        dash pattern=on 2pt off 1.6pt, line width=0.6pt] (0,0) rectangle (#1,#2);
  \node[ttl, anchor=north west] at (0.20,#2-0.14) {#4};}
